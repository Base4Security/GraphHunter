% GraphHunter Architecture — LaTeX preamble

\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern}

\usepackage[a4paper,margin=2.4cm]{geometry}

\usepackage{amsmath, amssymb, amsthm}
\usepackage{mathtools}

% Pseudocode — algorithm2e is more flexible for CLRS-style than algpseudocode.
% `ruled` = horizontal bars above and below; `linesnumbered` = 1. 2. ... gutter.
\usepackage[ruled,linesnumbered,algochapter]{algorithm2e}
\SetKwComment{Comment}{$\triangleright$ }{}
\SetKwFor{For}{for}{do}{end}
\SetKwFor{While}{while}{do}{end}
\SetKwIF{If}{ElseIf}{Else}{if}{then}{else if}{else}{end}
\SetKwRepeat{Repeat}{repeat}{until}
% Control-flow keywords we use as bare verbs inside pseudocode.
\SetKw{KwBreak}{break}
\SetKw{KwContinue}{continue}
\SetKw{KwTrue}{true}
\SetKw{KwFalse}{false}
% Aliases matching CLRS-ish capitalisation.
\newcommand{\Break}{\KwBreak}
\newcommand{\Continue}{\KwContinue}

% Inline code + short snippets.
\usepackage{listings}
\usepackage{xcolor}
\definecolor{rustcomment}{rgb}{0.35,0.45,0.45}
\definecolor{rustkw}{rgb}{0.55,0.10,0.55}
\definecolor{ruststr}{rgb}{0.10,0.45,0.10}
\lstdefinelanguage{Rust}{
    morekeywords={fn,let,mut,pub,struct,enum,impl,trait,where,if,else,for,while,
                  loop,match,return,use,mod,const,static,self,Self,as,in,ref,move,
                  async,await,dyn,unsafe,crate,super,extern,type,break,continue},
    morecomment=[l]{//},
    morecomment=[s]{/*}{*/},
    morestring=[b]",
    morestring=[b]',
}

    basicstyle=\ttfamily\small,

    commentstyle=\color{rustcomment}\itshape,

    breaklines=true,
    tabsize=4,
}
\lstdefinestyle{rust}{language=Rust}

    commentstyle=\color{rustcomment}\itshape,

% Macros.
\newcommand{\Ord}[1]{\ensuremath{O(#1)}}
\newcommand{\Thet}[1]{\ensuremath{\Theta(#1)}}
% Accept underscores, ampersands, and similar "special" characters
% verbatim inside \filepath and \code. `\detokenize` reads the
% argument as a token stream without expanding or evaluating math
% mode, so \code{foo_bar} renders as expected.
\newcommand{\filepath}[1]{\texttt{\small\detokenize{#1}}}
\newcommand{\code}[1]{\texttt{\detokenize{#1}}}
\newcommand{\StrId}{\ensuremath{\mathsf{StrId}}}
\newcommand{\Compact}{\ensuremath{\mathsf{CompactRelation}}}
\newcommand{\Graph}{\ensuremath{\mathsf{InMemoryGraph}}}
\newcommand{\Scorer}{\ensuremath{\mathsf{AnomalyScorer}}}

% Slightly looser Lines per algorithm environment so long pseudocode doesn't
% overflow pages.
\makeatletter
\renewcommand{\@algocf@pre@ruled}{\hrule height.8pt depth0pt \kern2pt}
\renewcommand{\@algocf@post@ruled}{\kern2pt\hrule height.8pt\relax}
\makeatother

% Part / chapter heading format tweaks — CLRS uses big Part pages.
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\huge\bfseries}
  {\chaptertitlename~\thechapter}
  {20pt}
  {\Huge}
